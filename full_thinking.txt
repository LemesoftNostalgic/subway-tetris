const AudioSys={
  ctx:null, sfxOn:true, musOn:true, _mTimer:null, _step:0, _nextT:0,
  init(){
    if(this.ctx) { if(this.ctx.state==='suspended') this.ctx.resume(); return; }
    try{
      const C=window.AudioContext||window.webkitAudioContext; this.ctx=new C();
    }catch(e){ return; }
    this.master=this.ctx.createGain(); this.master.gain.value=.9; this.master.connect(this.ctx.destination);
    this.sfxB=this.ctx.createGain(); this.sfxB.gain.value=.6; this.sfxB.connect(this.master);
    this.musB=this.ctx.createGain(); this.musB.gain.value=.30; this.musB.connect(this.master);
    this._startMusic();
  },
  tone(f,d,o={}){
    if(!this.ctx) return;
    const {type='square',v=.25,dest='sfx',slide=0,when=0}=o;
    const bus=dest==='mus'?this.musB:this.sfxB;
    if(dest==='mus'&&!this.musOn) return;
    if(dest==='sfx'&&!this.sfxOn) return;
    const t=this.ctx.currentTime+when;
    const osc=this.ctx.createOscillator(), g=this.ctx.createGain();
    osc.type=type; osc.frequency.setValueAtTime(f,t);
    if(slide) osc.frequency.exponentialRampToValueAtTime(Math.max(30,f+slide),t+d);
    g.gain.setValueAtTime(.0001,t);
    g.gain.exponentialRampToValueAtTime(v,t+.012);
    g.gain.exponentialRampToValueAtTime(.0001,t+d);
    osc.connect(g); g.connect(bus);
    osc.start(t); osc.stop(t+d+.06);
  },
  noise(d,o={}){
    if(!this.ctx) return;
    const {v=.3,dest='sfx',freq=1000,freqEnd=null,when=0}=o;
    const bus=dest==='mus'?this.musB:this.sfxB;
    if((dest==='mus'&&!this.musOn)||(dest==='sfx'&&!this.sfxOn)) return;
    const t=this.ctx.currentTime+when;
    const len=Math.max(1,Math.floor(this.ctx.sampleRate*d));
    const buf=this.ctx.createBuffer(1,len,this.ctx.sampleRate);
    const data=buf.getChannelData(0);
    for(let i=0;i<len;i++) data[i]=Math.random()*2-1;
    const src=this.ctx.createBufferSource(); src.buffer=buf;
    const f=this.ctx.createBiquadFilter(); f.type='lowpass'; f.frequency.setValueAtTime(freq,t);
    if(freqEnd) f.frequency.exponentialRampToValueAtTime(freqEnd,t+d);
    const g=this.ctx.createGain();
    g.gain.setValueAtTime(v,t); g.gain.exponentialRampToValueAtTime(.0001,t+d);
    src.connect(f); f.connect(g); g.connect(bus);
    src.start(t); src.stop(t+d);
  },

function collide(m,x,y){
  for(let cy=0;cy<m.length;cy++)for(let cx=0;cx<m[cy].length;cx++){
    if(!m[cy][cx]) continue;
    const bx=x+cx, by=y+cy;
    if(bx<0||bx>=COLS||by>=ROWS) return true;
    if(by>=0&&board[by][bx]) return true;
  }
  return false;
}
function rot(dir,m){
  const n=m.length;
  return dir>0 ? m.map((row,y)=>row.map((_,x)=>m[n-1-x][y]))
               : m.map((row,y)=>row.map((_,x)=>m[x][n-1-y]));
}

function updatePlaying(dt){
  const prev=timeLeft;
  timeLeft-=dt/1000;
  if(timeLeft<=0){ timeLeft=0; updateHUD(); gameOver('time'); return; }
  if(prev>20&&timeLeft<=20) sfx.warn();
  if(timeLeft<=10){ const s=Math.ceil(timeLeft); if(s!==lastTickSec){ lastTickSec=s; sfx.tick(); } }

  if(clearing){
    clearT+=dt;
    if(clearT>=340){
      removeCleared(); clearing=false;
      if(isWin()) startTrain(); else spawn();
      updateGoalPanel();
    }
    return;
  }
  if(!piece) return;

  handleRepeat('left',dt); handleRepeat('right',dt);
  if(keys.down){
    softT+=dt; let stepped=false;
    while(softT>=45){ softT-=45; if(move(0,1)){ score+=1; stepped=true; } else break; }
    if(stepped) sfx.soft();
  } else {
    dropT+=dt;
    if(dropT>=dropInterval()){ dropT=0; move(0,1); }
  }
  if(!canDown()){ lockT+=dt; if(lockT>=240) lockPiece(); }
  else lockT=0;
}
function handleRepeat(dir,dt){

      repT[dir]-=55;
      if(!move(dir==='left'?-1:1,0)){ repT[dir]=0; break; }
    }
  }
}

**How it plays**

- **The goal, not lines:** each level places an amber "PLATFORM" line a bit above the middle (its height shifts between levels: rows 6–10). You win by making every column's top land *exactly* on that line — holes below are fine, but any block poking above blocks the train. When the surface is flat, the signal flips green, the Line-7 train horns, crosses the bridge over the gap, and the whole platform cascades away for a level bonus plus a time bonus.
- **One subtle rule handled for you:** the flat surface row is technically a "full line" in Tetris terms. The game intercepts that moment — if it completes with nothing above it, it's a win, not a line clear — so the goal is actually achievable.
- **Difficulty curve:** Level 1 gives 2:30, then the clock tightens by 12s per level (floor 55s), gravity speeds up, and the target line moves.
- **Feel & juice:** 7-bag randomizer, ghost piece, DAS auto-repeat, hard-drop dust, screen shake, line-clear flashes, train sparks, confetti, per-column progress pips with an "obstruction" warning, chiptune music + full SFX (all synthesized in Web Audio, no files), pause/mute, touch controls on mobile, and a persistent station record in `localStorage`.

**Controls:** ← → move · ↑/X rotate · Z counter-rotate · ↓ soft drop · Space hard drop · P pause · M sound · Enter/R to start or reboard.
